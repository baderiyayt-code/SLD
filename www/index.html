Create table to sync offline app state automatically CREATE TABLE survey_data ( user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, data JSONB NOT NULL, updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW() ); -- 2. Enable Row Level Security (RLS) for absolute privacy ALTER TABLE survey_data ENABLE ROW LEVEL SECURITY; -- 3. Policy: User can only see their own survey data CREATE POLICY "Users can view own data" ON survey_data FOR SELECT USING (auth.uid() = user_id); -- 4. Policy: User can insert/update their own survey data CREATE POLICY "Users can insert/update own data" ON survey_data FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
